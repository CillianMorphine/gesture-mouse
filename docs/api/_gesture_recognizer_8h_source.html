42/**
43 * @brief Перелік усіх розпізнаваних типів жестів.
44 *
45 * @details
46 * Відповідність жестів → дій миші:
47 *
48 * | Значення | Жест | Дія миші |
49 * |----------|------|----------|
50 * | NONE | Рука не виявлена | — |
51 * | MOVE | Відкрита долоня (4+ пальців) | Переміщення курсору |
52 * | LEFT_CLICK | Вказівний палець | Клік лівою кнопкою |
53 * | RIGHT_CLICK | Два пальці (V) | Клік правою кнопкою |
54 * | SCROLL_UP | Рух угору з долонею | Прокрутка вгору |
55 * | SCROLL_DOWN | Рух вниз з долонею | Прокрутка вниз |
56 * | DRAG | Стиснутий кулак у русі | Перетягування |
57 *
58 * @note Значення enum є стабільними — їх можна зберігати у конфіг-файлах.
59 */
60enum class GestureType : uint8_t {
61 NONE = 0, ///< Жест не виявлено або рука поза кадром
62 MOVE = 1, ///< Переміщення курсору (відкрита долоня)
63 LEFT_CLICK = 2, ///< Клік лівою кнопкою (1 палець)
64 RIGHT_CLICK = 3, ///< Клік правою кнопкою (2 пальці — V)
65 SCROLL_UP = 4, ///< Прокрутка вгору
66 SCROLL_DOWN = 5, ///< Прокрутка вниз
67 DRAG = 6, ///< Перетягування (кулак + рух)
68};

70/**
71 * @brief Результат розпізнавання одного кадру відеопотоку.
72 *
73 * @details
74 * Структура є **value type** (копіюється дешево).
75 * Координати нормалізовані до діапазону [0.0, 1.0] відносно
76 * розмірів кадру камери — MouseController перетворює їх у
77 * піксельні координати екрана.
78 *
79 * @par Приклад використання:
80 * @code{.cpp}
81 * GestureResult result = recognizer.process(frame);
82 * if (result.type == GestureType::MOVE && result.confidence > 0.7F) {
83 * mouseCtrl.applyGesture(result);
84 * }
85 * @endcode
86 */
87struct GestureResult {
88 GestureType type{GestureType::NONE}; ///< Тип розпізнаного жесту
89 float handX{0.0F}; ///< Нормалізована X-координата центру руки [0..1]
90 float handY{0.0F}; ///< Нормалізована Y-координата центру руки [0..1]
91 float confidence{0.0F}; ///< Впевненість класифікатора [0..1], 1.0 = максимум
92
93 /**
94 * @brief Повертає рядкову назву типу жесту.
95 *
96 * @details Зручно для логування та debug-виводу:
97 * @code{.cpp}
98 * LOG_DEBUG("Gesture: " + result.typeName());
99 * @endcode
100 *
101 * @return Рядок у верхньому регістрі, наприклад `"LEFT_CLICK"`,
102 * або `"NONE"` якщо жест не виявлено.
103 */
104 [[nodiscard]] std::string typeName() const;
105};

112/**
113 * @brief Фасад модуля розпізнавання — координує HandTracker і GestureClassifier.
114 *
115 * @details
116 * GestureRecognizer є **єдиною публічною точкою входу** до підсистеми CV.
117 * Application взаємодіє лише з цим класом, не знаючи деталей реалізації
118 * HandTracker або GestureClassifier.
119 *
120 * ### Патерн Facade
121 * ```
122 * Application
123 * │
124 * └──► GestureRecognizer::process(frame)
125 * ├──► HandTracker::detect(frame) → HandData
126 * └──► GestureClassifier::classify(hand) → GestureResult
127 * ```
128 *
129 * ### Налаштування через ConfigManager
130 * | Ключ | Тип | За замовч. | Опис |
131 * |------|-----|------------|------|
132 * | `gesture.confidence` | float | 0.6 | Мін. впевненість для підтвердження |
133 * | `camera.index` | int | 0 | Індекс камери для OpenCV |
134 * | `debug.show_window` | bool | false | Показувати debug-кадр |
135 */

163 /**
164 * @brief Обробляє один кадр і повертає розпізнаний жест.
165 *
166 * @details
167 * **Алгоритм обробки:**
168 * 1. `HandTracker::detect(frame)` → знаходить контур руки через HSV-маску
169 * 2. Якщо рука не виявлена — повертає `GestureResult{GestureType::NONE}`
170 * 3. `GestureClassifier::classify(handData)` → визначає тип жесту
171 * 4. Якщо ввімкнено debug-вікно — накладає landmarks на кадр
172 *
173 * **Продуктивність:** типовий час виконання < 5 мс на CPU (Intel i5).
174 *
175 * @param frame Кадр BGR з камери (CV_8UC3). Не модифікується.
176 *
177 * @return GestureResult з типом жесту, координатами та впевненістю.
178 * Якщо frame порожній — повертає NONE без помилок.
179 *
180 * @note Метод **не є потокобезпечним** — викликати тільки з одного потоку.
181 *
182 * @par Приклад:
183 * @code{.cpp}
184 * cv::VideoCapture cap(0);
185 * cv::Mat frame;
186 * while (cap.read(frame)) {
187 * auto result = recognizer.process(frame);
188 * if (result.type != GestureType::NONE)
189 * LOG_INFO("Gesture: " + result.typeName());
190 * }
191 * @endcode
192 */
193 [[nodiscard]] GestureResult process(const cv::Mat& frame);
